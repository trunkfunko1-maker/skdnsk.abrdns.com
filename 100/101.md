<!-- 🚀 通用视频预加载防卡顿脚本 -->
<script>
document.addEventListener('DOMContentLoaded', function() {
    const wrappers = document.querySelectorAll('.video-wrapper');
    wrappers.forEach(wrapper => {
        const video = wrapper.querySelector('video');
        const loader = wrapper.querySelector('.loading-overlay');
        const loaderText = wrapper.querySelector('.loader-text');
        let isReady = false;

        video.addEventListener('progress', function() {
            if (video.buffered.length > 0) {
                const bufferedEnd = video.buffered.end(video.buffered.length - 1);
                const duration = video.duration;
                if (duration > 0 && (bufferedEnd / duration) >= 0.15 && !isReady) {
                    isReady = true;
                    loader.style.display = 'none';
                }
            }
        });

        video.addEventListener('canplaythrough', function() {
            isReady = true;
            loader.style.display = 'none';
        });

        video.addEventListener('play', function(e) {
            if (!isReady) {
                video.pause();
                loader.style.display = 'flex';
                loaderText.innerText = '网络较慢，正在拼命缓冲中...';
                video.play().catch(() => {});
            }
        });

        video.addEventListener('waiting', function() {
            loader.style.display = 'flex';
            loaderText.innerText = '网络波动，正在重新缓冲...';
        });

        video.addEventListener('playing', function() {
            loader.style.display = 'none';
        });

        if (video.readyState >= 3) {
            isReady = true;
            loader.style.display = 'none';
        }
    });
});
</script>
